cmake_minimum_required(VERSION 4.4.3)
project(CST VERSION 1.0.0 LANGUAGES CXX RC)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CST requires Windows x64 and MSVC 2022 v143. Portable unit tests use cmake -S tests separately.")
endif()
if(NOT CMAKE_GENERATOR STREQUAL "Ninja Multi-Config")
  message(FATAL_ERROR "Use Ninja Multi-Config through CMakePresets.json")
endif()
if(MSVC_VERSION LESS 1930 OR MSVC_VERSION GREATER_EQUAL 1950)
  message(FATAL_ERROR "Use the MSVC 2022 v143 toolset")
endif()
if(NOT CMAKE_SYSTEM_VERSION MATCHES "^10\\.0\\.26100(\\.0)?$")
  message(FATAL_ERROR "Windows SDK 10.0.26100 is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Gui Widgets Network Test)
get_target_property(qt_core_type Qt6::Core TYPE)
if(NOT qt_core_type STREQUAL "SHARED_LIBRARY")
  message(FATAL_ERROR "CST requires dynamically linked Qt")
endif()
enable_testing()
option(CST_PRODUCTION_SIGNING "Require production signing material" OFF)
set(CST_IFW_ROOT "$ENV{CST_IFW_ROOT}" CACHE PATH "Qt Installer Framework 4.11.0 root")
if(CST_PRODUCTION_SIGNING)
  foreach(variable CST_SIGN_PFX CST_SIGN_PFX_PASSWORD CST_SIGN_TIMESTAMP_URL)
    if("$ENV{${variable}}" STREQUAL "")
      message(FATAL_ERROR "Production packaging requires ${variable}")
    endif()
  endforeach()
endif()

function(cst_owned target)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
endfunction()
add_library(cst_domain STATIC src/domain/Configuration.cpp src/domain/Runtime.cpp)
target_include_directories(cst_domain PUBLIC src)
target_link_libraries(cst_domain PUBLIC Qt6::Core Qt6::Network)
cst_owned(cst_domain)
add_library(cst_application STATIC
  src/application/ProjectConfigService.cpp src/application/ReadinessService.cpp
  src/application/PortReclaimService.cpp src/application/SourceSyncService.cpp
  src/application/TaskSupervisor.cpp src/application/OperationQueue.cpp
  src/application/ProjectRuntimeService.cpp src/application/LogService.cpp
  src/application/DiagnosticExportService.cpp src/infrastructure/common/SystemClock.cpp)
target_link_libraries(cst_application PUBLIC cst_domain)
cst_owned(cst_application)
add_library(cst_platform_windows STATIC
  src/infrastructure/windows/WindowsPlatform.cpp src/infrastructure/windows/WindowsProcessRunner.cpp
  src/infrastructure/windows/WindowsPortManager.cpp src/infrastructure/windows/WindowsSession.cpp)
target_link_libraries(cst_platform_windows PUBLIC cst_application PRIVATE Advapi32 Iphlpapi Ws2_32 Shell32 User32 Ole32)
cst_owned(cst_platform_windows)
add_library(cst_ui STATIC src/ui/MainWindow.cpp src/ui/widgets/SchemaEditor.cpp
  src/ui/pages/user/UserPage.cpp src/ui/pages/admin/AdminPage.cpp)
target_link_libraries(cst_ui PUBLIC cst_application Qt6::Widgets)
cst_owned(cst_ui)
qt_add_executable(cst WIN32 src/app/main.cpp resources/cst.rc)
target_link_libraries(cst PRIVATE cst_ui cst_platform_windows)
target_include_directories(cst PRIVATE resources)
target_link_options(cst PRIVATE /MANIFEST:NO)
qt_add_resources(cst "configuration" PREFIX "/config" BASE config FILES config/cst-project.schema.json config/cst-project.example.json)
cst_owned(cst)
add_executable(cst-git-askpass helpers/git-askpass/main.cpp)
target_link_libraries(cst-git-askpass PRIVATE cst_platform_windows)
cst_owned(cst-git-askpass)
add_executable(cst-signal-helper helpers/signal-helper/main.cpp)
set_target_properties(cst-signal-helper PROPERTIES AUTOMOC OFF)
cst_owned(cst-signal-helper)

foreach(test Domain Config Service)
  string(TOLOWER "${test}" lower)
  add_executable(cst-unit-${lower} tests/unit/${test}Tests.cpp)
  target_link_libraries(cst-unit-${lower} PRIVATE cst_application Qt6::Test)
  target_compile_definitions(cst-unit-${lower} PRIVATE CST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  cst_owned(cst-unit-${lower})
  add_test(NAME unit.${lower} COMMAND cst-unit-${lower})
endforeach()

add_executable(cst-unit-ui tests/unit/UiTests.cpp)
target_link_libraries(cst-unit-ui PRIVATE cst_ui Qt6::Test)
target_compile_definitions(cst-unit-ui PRIVATE CST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
cst_owned(cst-unit-ui)
foreach(scale 1.25 1.5 2)
  add_test(NAME ui.${scale} COMMAND cst-unit-ui)
  set_tests_properties(ui.${scale} PROPERTIES ENVIRONMENT "QT_SCALE_FACTOR=${scale}" TIMEOUT 30)
endforeach()

install(TARGETS cst cst-git-askpass cst-signal-helper RUNTIME DESTINATION .)
add_executable(cst-integration-git tests/integration/GitSyncTests.cpp)
target_link_libraries(cst-integration-git PRIVATE cst_platform_windows Qt6::Test)
cst_owned(cst-integration-git)
add_test(NAME integration.git COMMAND cst-integration-git)
set_tests_properties(integration.git PROPERTIES TIMEOUT 120)
add_executable(cst-test-port-owner tests/fixtures/PortOwner.cpp)
target_link_libraries(cst-test-port-owner PRIVATE Qt6::Core Qt6::Network)
cst_owned(cst-test-port-owner)
add_executable(cst-integration-windows tests/integration/WindowsProcessTests.cpp)
target_link_libraries(cst-integration-windows PRIVATE cst_platform_windows Qt6::Test)
cst_owned(cst-integration-windows)
add_test(NAME integration.windows COMMAND cst-integration-windows)
set_tests_properties(integration.windows PROPERTIES LABELS admin TIMEOUT 120)
install(DIRECTORY LICENSES/ DESTINATION LICENSES)
install(FILES config/cst-project.schema.json config/cst-project.example.json DESTINATION config)
get_target_property(qmake_executable Qt6::qmake IMPORTED_LOCATION)
get_filename_component(qt_bin "${qmake_executable}" DIRECTORY)
set(package_root "${CMAKE_BINARY_DIR}/installer")
set(package_data "${package_root}/packages/com.cst.app/data")
add_custom_target(package-installer
  COMMAND "${CMAKE_COMMAND}" -DCST_SOURCE=${CMAKE_SOURCE_DIR} -DCST_BUILD=${CMAKE_BINARY_DIR}
    -DCST_QT_BIN=${qt_bin} -DCST_IFW_ROOT=${CST_IFW_ROOT} -DCST_PRODUCTION_SIGNING=${CST_PRODUCTION_SIGNING}
    -P "${CMAKE_SOURCE_DIR}/cmake/Package.cmake"
  DEPENDS cst cst-git-askpass cst-signal-helper
  USES_TERMINAL VERBATIM)
